  const token = getStoredToken();
  const headers = { 'Content-Type': 'application/json' };
  if (token && isValidToken(token)) headers.Authorization = `Bearer ${token}`;

    throw error;
  }
};

import axios from 'axios';
import { API, AUTH, HTTP_STATUS } from '../utils/constants.js';
import { getStoredToken, isValidToken, removeToken } from '../utils/jwt.js';

/**
 * Authentication Service
 * Handles all authentication-related API calls and token management
 */

// Create axios instance with default config
const authApi = axios.create({
  baseURL: API.BASE_URL,
  timeout: API.TIMEOUT,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Request interceptor to add auth token
authApi.interceptors.request.use(
  (config) => {
    const token = getStoredToken();
    if (token && isValidToken(token)) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// Response interceptor to handle auth errors
authApi.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === HTTP_STATUS.UNAUTHORIZED) {
      // Token expired or invalid, remove it and redirect to login
      removeToken();
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

/**
 * Login user with username and password
 * @param {object} credentials - Login credentials
 * @param {string} credentials.username - Username
 * @param {string} credentials.password - Password
 * @returns {Promise<object>} - Login response with token
 */
export const login = async (credentials) => {
  try {
    const response = await authApi.post(`${API.ENDPOINTS.AUTH}/login`, credentials);
    
    if (response.data.success && response.data.data?.token) {
      return {
        success: true,
        token: response.data.data.token,
        message: response.data.message || 'Login successful',
      };
    } else {
      return {
        success: false,
        error: 'INVALID_RESPONSE',
        message: 'Invalid response from server',
      };
    }
  } catch (error) {
    console.error('Login error:', error);
    
    if (error.response?.data) {
      return {
        success: false,
        error: error.response.data.error || 'LOGIN_FAILED',
        message: error.response.data.message || 'Login failed',
      };
    }
    
    return {
      success: false,
      error: 'NETWORK_ERROR',
      message: 'Network error. Please check your connection.',
    };
  }
};

/**
 * Logout user (client-side token removal)
 * @returns {Promise<object>} - Logout result
 */
export const logout = async () => {
  try {
    removeToken();
    return {
      success: true,
      message: 'Logged out successfully',
    };
  } catch (error) {
    console.error('Logout error:', error);
    return {
      success: false,
      error: 'LOGOUT_ERROR',
      message: 'Error during logout',
    };
  }
};

/**
 * Verify current token validity
 * @returns {Promise<object>} - Token verification result
 */
export const verifyToken = async () => {
  try {
    const token = getStoredToken();
    
    if (!token || !isValidToken(token)) {
      return {
        success: false,
        error: 'INVALID_TOKEN',
        message: 'No valid token found',
      };
    }
    
    // You could add a backend endpoint to verify token if needed
    // For now, we'll rely on client-side validation
    return {
      success: true,
      message: 'Token is valid',
    };
  } catch (error) {
    console.error('Token verification error:', error);
    return {
      success: false,
      error: 'VERIFICATION_ERROR',
      message: 'Error verifying token',
    };
  }
};

/**
 * Check if user has permission for specific role
 * @param {string} requiredRole - Required role (ADMIN/OWNER)
 * @returns {Promise<object>} - Permission check result
 */
export const checkPermission = async (requiredRole) => {
  try {
    const verification = await verifyToken();
    
    if (!verification.success) {
      return verification;
    }
    
    const token = getStoredToken();
    const decoded = token ? JSON.parse(atob(token.split('.')[1])) : null;
    
    if (!decoded || decoded.role !== requiredRole) {
      return {
        success: false,
        error: 'INSUFFICIENT_PERMISSIONS',
        message: 'Insufficient permissions for this action',
      };
    }
    
    return {
      success: true,
      message: 'Permission granted',
    };
  } catch (error) {
    console.error('Permission check error:', error);
    return {
      success: false,
      error: 'PERMISSION_ERROR',
      message: 'Error checking permissions',
    };
  }
};

/**
 * Get user profile information
 * This would typically call a backend endpoint to get user details
 * For now, it extracts info from the stored token
 */
export const getUserProfile = async () => {
  try {
    const verification = await verifyToken();
    
    if (!verification.success) {
      return verification;
    }
    
    const token = getStoredToken();
    const decoded = token ? JSON.parse(atob(token.split('.')[1])) : null;
    
    if (!decoded) {
      return {
        success: false,
        error: 'INVALID_TOKEN',
        message: 'Could not decode user information',
      };
    }
    
    return {
      success: true,
      data: {
        id: decoded.userId,
        username: decoded.username,
        role: decoded.role,
        subject: decoded.sub,
        issuedAt: new Date(decoded.iat * 1000).toISOString(),
        expiresAt: new Date(decoded.exp * 1000).toISOString(),
      },
      message: 'User profile retrieved successfully',
    };
  } catch (error) {
    console.error('Get user profile error:', error);
    return {
      success: false,
      error: 'PROFILE_ERROR',
      message: 'Error retrieving user profile',
    };
  }
};

function App() {
  // Function to determine where to redirect authenticated users
  const getAuthenticatedRedirect = () => {
    if (isAuthenticated()) {
      const token = getStoredToken();
      return getDashboardRoute(token);
    }
    return '/login';
  };

  return (
    <Router>
      <div className="App">
        <Routes>
          {/* Public Routes (Guest Only) */}
          <Route
            path="/login"
            element={
              <GuestRoute>
                <Login />
              </GuestRoute>
            }
          />

          {/* Protected Admin Routes with nested layout */}
          <Route
            path="/admin"
            element={

              </OwnerRoute>
            }
          >

            <Route path="admin-panel-viewer" element={<OwnerAdminPanelViewer />} />

          {/* Root Route - Redirect based on auth status */}
          <Route
            path="/"
            element={<Navigate to={getAuthenticatedRedirect()} replace />}
          />

          <Route path="/admin/dashboard" element={<Navigate to="/admin" replace />} />
          <Route path="/owner/dashboard" element={<Navigate to="/owner" replace />} />

          {/* Catch-all Route - Redirect to appropriate dashboard or login */}
          <Route
            path="*"
            element={<Navigate to={getAuthenticatedRedirect()} replace />}
          />
        </Routes>
      </div>
    </Router>
  );

import React, { useEffect, useState } from 'react';
import { Navigate, useLocation } from 'react-router-dom';
import { isAuthenticated, getCurrentUser } from '../utils/jwt.js';
import { AUTH } from '../utils/constants.js';

/**
 * ProtectedRoute Component
 * Wraps components that require authentication and/or specific roles
 */
const ProtectedRoute = ({ 
  children, 
  requiredRole = null, 
  redirectTo = '/login' 
}) => {
  const [isChecking, setIsChecking] = useState(true);
  const [hasAccess, setHasAccess] = useState(false);
  const location = useLocation();

  useEffect(() => {
    const checkAccess = async () => {
      try {
        // Check if user is authenticated
        if (!isAuthenticated()) {
          setHasAccess(false);
          setIsChecking(false);
          return;
        }

        // If no specific role required, just check authentication
        if (!requiredRole) {
          setHasAccess(true);
          setIsChecking(false);
          return;
        }

        // Check if user has required role
        const currentUser = getCurrentUser();
        if (!currentUser) {
          setHasAccess(false);
          setIsChecking(false);
          return;
        }

        const userHasRole = currentUser.role === requiredRole;
        setHasAccess(userHasRole);
        setIsChecking(false);
      } catch (error) {
        console.error('Access check error:', error);
        setHasAccess(false);
        setIsChecking(false);
      }
    };

    checkAccess();
  }, [requiredRole]);

  // Show loading spinner while checking access
  if (isChecking) {
    return (
      <div className="d-flex justify-content-center align-items-center min-vh-100">
        <div className="text-center">
          <div className="spinner-border text-primary mb-3" role="status">
            <span className="visually-hidden">Loading...</span>
          </div>
          <p className="text-muted">Verifying access...</p>
        </div>
      </div>
    );
  }

  // Redirect if no access
  if (!hasAccess) {
    return (
      <Navigate 
        to={redirectTo} 
        state={{ from: location }} 
        replace 
      />
    );
  }

  // Render protected content
  return children;
};

/**
 * AdminRoute Component
 * Specifically for admin-only routes
 */
export const AdminRoute = ({ children, redirectTo = '/login' }) => {

    <ProtectedRoute requiredRole={AUTH.ROLES.ADMIN} redirectTo={redirectTo}>
      {children}
    </ProtectedRoute>
  );
};

export const AuthRoute = ({ children, redirectTo = '/login' }) => {

    </ProtectedRoute>

};

/**
 * GuestRoute Component
 * For unauthenticated users only (redirects authenticated users to their dashboard)
 */
export const GuestRoute = ({ children }) => {
  const [isChecking, setIsChecking] = useState(true);
  const [isAuth, setIsAuth] = useState(false);
  const [dashboardRoute, setDashboardRoute] = useState('/');

  useEffect(() => {
    checkAuthStatus();
  }, []);

  const checkAuthStatus = async () => {
    try {
      const authenticated = isAuthenticated();
      setIsAuth(authenticated);

      if (authenticated) {
        const currentUser = getCurrentUser();
        if (currentUser) {
          const route = currentUser.role === AUTH.ROLES.ADMIN 
            ? '/admin/dashboard' 
            : '/owner/dashboard';
          setDashboardRoute(route);
        }
      }
      
      setIsChecking(false);
    } catch (error) {
      console.error('Auth status check error:', error);
      setIsAuth(false);
      setIsChecking(false);
    }
  };

  // Show loading while checking
  if (isChecking) {
    return (
      <div className="d-flex justify-content-center align-items-center min-vh-100">
        <div className="spinner-border text-primary" role="status">
          <span className="visually-hidden">Loading...</span>
        </div>
      </div>
    );
  }

  // Redirect authenticated users to their dashboard
  if (isAuth) {
    return <Navigate to={dashboardRoute} replace />;
  }

  // Show content for unauthenticated users
  return children;
};

import { createAsyncThunk, createSlice } from '@reduxjs/toolkit';
import { login as loginApi, logout as logoutApi, getUserProfile } from '../../api/authService.js';
import { getStoredToken, storeToken, removeToken } from '../../utils/jwt.js';

const initialState = {
  token: getStoredToken() || null,
  user: null,
  loading: false,
  error: null,
};

export const login = createAsyncThunk('auth/login', async (credentials, { rejectWithValue }) => {
  try {
    const res = await loginApi(credentials);
    if (!res.success) return rejectWithValue(res.message || 'Login failed');
    return res.token;
  } catch (e) {
    return rejectWithValue(e?.message || 'Login failed');
  }
});

export const fetchProfile = createAsyncThunk('auth/fetchProfile', async (_, { rejectWithValue }) => {
  try {
    const res = await getUserProfile();
    if (!res.success) return rejectWithValue(res.message || 'Failed to load profile');
    return res.data;
  } catch (e) {
    return rejectWithValue(e?.message || 'Failed to load profile');
  }
});

export const logout = createAsyncThunk('auth/logout', async () => {
  await logoutApi();
  return true;
});

const authSlice = createSlice({
  name: 'auth',
  initialState,
  reducers: {},
  extraReducers: (builder) => {
    builder
      .addCase(login.pending, (state) => {
        state.loading = true; state.error = null;
      })
      .addCase(login.fulfilled, (state, action) => {
        state.loading = false; state.token = action.payload; state.error = null;
        storeToken(action.payload);
      })
      .addCase(login.rejected, (state, action) => {
        state.loading = false; state.error = action.payload || 'Login failed';
      })
      .addCase(fetchProfile.pending, (state) => {
        state.loading = true; state.error = null;
      })
      .addCase(fetchProfile.fulfilled, (state, action) => {
        state.loading = false; state.user = action.payload;
      })
      .addCase(fetchProfile.rejected, (state, action) => {
        state.loading = false; state.error = action.payload || 'Failed to load profile';
      })
      .addCase(logout.fulfilled, (state) => {
        state.token = null; state.user = null; removeToken();
      });
  }
});

export default authSlice.reducer;

// Helper sub-component: Focused Target Mode (WebSocket + uninstall control toggles)
const TargetModeSection = ({ deviceOnline=false }) => {
  const [wsConnected, setWsConnected] = React.useState(false);
  const [connecting, setConnecting] = React.useState(false);
  const [lastPingMs, setLastPingMs] = React.useState(null);
  const [pingHistory, setPingHistory] = React.useState([]); // last few pings
  const [enforceUninstallBlock, setEnforceUninstallBlock] = React.useState(true);
  const [focusedPackages, setFocusedPackages] = React.useState(['pkg.whatsapp','com.paytm.app']);
  const [newPkg, setNewPkg] = React.useState('');
  const [showErrModal, setShowErrModal] = React.useState(false);
  const [errReason, setErrReason] = React.useState('');
  const wsRef = React.useRef(null);
  const pingTimerRef = React.useRef(null);

  const addPkg = () => { if (!newPkg.trim()) return; setFocusedPackages(p=> p.includes(newPkg.trim())?p:[...p,newPkg.trim()]); setNewPkg(''); };
  const removePkg = (pkg) => setFocusedPackages(p=>p.filter(x=>x!==pkg));

  const cleanupWs = React.useCallback(() => {
    if (pingTimerRef.current) { clearInterval(pingTimerRef.current); pingTimerRef.current = null; }
  if (wsRef.current) { try { wsRef.current.close(); } catch { /* ignore close errors */ } wsRef.current = null; }
  }, []);

  React.useEffect(() => () => cleanupWs(), [cleanupWs]);

  const attemptConnect = () => {
    if (!deviceOnline || wsConnected || connecting) return;
    setConnecting(true);
    // Simulate server requesting Android client for a session handshake
    // In real impl: POST /api/device/:id/ws-handshake -> returns temp ws url/token
    setTimeout(() => {
      const shouldSucceed = Math.random() > 0.25; // 75% success sample
      if (!shouldSucceed) {
        setConnecting(false);
        setErrReason('Device refused handshake or network timeout. Ensure agent service is running and permissions granted.');
        setShowErrModal(true);
        return;
      }
      // Fake WebSocket object for demo
      wsRef.current = { readyState: 1, close: ()=>{} };
      setWsConnected(true);
      setConnecting(false);
      // Start ping simulation
      pingTimerRef.current = setInterval(() => {
        const start = performance.now();
        // simulate latency 50-400ms
        const latency = 50 + Math.random()*350;
        setTimeout(() => {
          const ms = Math.round(performance.now() - start + (Math.random()*10));
          setLastPingMs(ms);
          setPingHistory(h => {
            const next = [...h, ms];
            if (next.length > 20) next.shift();
            return next;
          });
          // Randomly simulate drop
          if (Math.random() < 0.02) {
            cleanupWs();
            setWsConnected(false);
            setErrReason('Connection lost (simulated network drop).');
            setShowErrModal(true);
          }
        }, latency);
      }, 5000);
    }, 900); // handshake delay
  };

  const disconnect = () => {
    cleanupWs();
    setWsConnected(false);
    setLastPingMs(null);
    setPingHistory([]);
  };

  const avgPing = React.useMemo(() => {
    if (!pingHistory.length) return null;
    return Math.round(pingHistory.reduce((a,b)=>a+b,0)/pingHistory.length);
  }, [pingHistory]);

  const pingQuality = (ms) => {
    if (ms == null) return 'N/A';
    if (ms < 120) return 'Good';
    if (ms < 250) return 'Fair';
    return 'Poor';
  };

  return (
    <div className="row g-3">
      <div className="col-12 d-flex flex-wrap justify-content-between align-items-center gap-2 mb-2">
        <h6 className="mb-0">Focused Target Mode</h6>
        <div className="d-flex gap-2 align-items-center flex-wrap">
          <div className="small">Device: {deviceOnline ? <span className="text-success fw-semibold">Online</span> : <span className="text-danger fw-semibold">Offline</span>}</div>
          <div className="small">WS: {wsConnected ? <span className="text-success">Connected</span> : connecting ? <span className="text-warning">Connecting...</span> : <span className="text-muted">Idle</span>}</div>
          {lastPingMs!=null && <div className="small">Last Ping: <span className={lastPingMs<250? 'text-success': lastPingMs<400? 'text-warning':'text-danger'}>{lastPingMs}ms</span></div>}
          {avgPing!=null && <div className="small">Avg: {avgPing}ms ({pingQuality(avgPing)})</div>}
          <button
            className={`btn btn-sm ${wsConnected? 'btn-outline-danger' : 'btn-outline-primary'}`}
            disabled={connecting || (!wsConnected && !deviceOnline)}
            onClick={wsConnected ? disconnect : attemptConnect}
          >
            {connecting ? 'Connecting...' : wsConnected ? 'Disconnect' : 'Connect'}
          </button>
          <button className="btn btn-sm btn-outline-secondary" disabled={!wsConnected} onClick={()=>setLastPingMs(null)}>Clear Stats</button>
        </div>
      </div>
      <div className="col-12 col-lg-5">
        <div className="card h-100">
          <div className="card-body">
            <h6 className="fw-semibold mb-3">Focused Packages</h6>
            <div className="input-group input-group-sm mb-2">
              <input className="form-control" placeholder="Add package e.g. com.app" value={newPkg} onChange={e=>setNewPkg(e.target.value)} disabled={!deviceOnline} />
              <button className="btn btn-outline-primary" onClick={addPkg} disabled={!newPkg.trim() || !deviceOnline}>Add</button>
            </div>
            {focusedPackages.length===0 && <div className="text-muted small">No packages targeted.</div>}
            <ul className="list-group small">
              {focusedPackages.map(pkg => (
                <li key={pkg} className="list-group-item d-flex justify-content-between align-items-center">
                  <span>{pkg}</span>
                  <button className="btn btn-sm btn-outline-danger" onClick={()=>removePkg(pkg)} disabled={!deviceOnline}>Remove</button>
                </li>
              ))}
            </ul>
          </div>
          <div className="card-footer text-muted small">Sample (not persisted)</div>
        </div>
      </div>
      <div className="col-12 col-lg-7">
        <div className="card h-100">
          <div className="card-body">
            <h6 className="fw-semibold mb-3">Session / Control</h6>
            <div className="row g-3">
              <div className="col-12 col-md-6">
                <div className="border rounded p-2 h-100">
                  <div className="fw-semibold small mb-1">Connection</div>
                  <div className="small mb-2">State: {wsConnected ? <span className="text-success">Connected</span> : connecting ? <span className="text-warning">Connecting...</span> : <span className="text-danger">Disconnected</span>}</div>
                  <button
                    className={`btn btn-sm w-100 mb-1 ${wsConnected? 'btn-danger':'btn-primary'}`}
                    onClick={wsConnected ? disconnect : attemptConnect}
                    disabled={connecting || (!wsConnected && !deviceOnline)}
                  >
                    {connecting ? 'Connecting...' : wsConnected ? 'Disconnect' : 'Connect'}
                  </button>
                  {wsConnected && <div className="small text-success mb-1">Active session</div>}
                  {!deviceOnline && <div className="small text-danger mt-2">Device offline. Cannot start session.</div>}
                </div>
              </div>
              <div className="col-12 col-md-6">
                <div className="border rounded p-2 h-100">
                  <div className="fw-semibold small mb-1">Uninstall Guard</div>
                  <div className="small mb-2">Prevents removal of focused packages when ON.</div>
                  <button className="btn btn-sm btn-warning w-100 mb-2" onClick={()=>setEnforceUninstallBlock(s=>!s)} disabled={!deviceOnline}>
                    {enforceUninstallBlock? 'Disable Guard':'Enable Guard'}
                  </button>
                  <div className="form-check form-switch small">
                    <input className="form-check-input" type="checkbox" id="guardSwitch" checked={enforceUninstallBlock} onChange={()=>setEnforceUninstallBlock(s=>!s)} disabled={!deviceOnline} />

                  <div className="fw-semibold small mb-2">Activity Log (Sample)</div>

                    {[`[${new Date().toLocaleTimeString()}] UI loaded`,
                      deviceOnline? `[${new Date().toLocaleTimeString()}] Device reachable`:`[${new Date().toLocaleTimeString()}] Device offline`,
                      wsConnected?`[${new Date().toLocaleTimeString()}] WS handshake success`:`[${new Date().toLocaleTimeString()}] WS idle`,
                      lastPingMs!=null?`[${new Date().toLocaleTimeString()}] Last ping ${lastPingMs}ms`:'[--] No ping yet',

      const action = await dispatch(updateAdminPassword({ id: selected.id, newPassword }));
      if (updateAdminPassword.fulfilled.match(action)) {
        setShowEditPwd(false);
      } else {
        alert(action.payload || 'Failed to update password');
      }
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to update password');
    } finally {
      setLoading(false);
    }
  };

  const updateExpiry = async (e) => {
    e.preventDefault();
    if (!selected) return;
    try {
      setLoading(true);
      const expiryInDays = parseInt(newExpiryRef.current.value, 10);
      if (!expiryInDays || expiryInDays <= 0) return;
      const action = await dispatch(updateAdminExpiryThunk({ id: selected.id, expiryInDays }));
      if (updateAdminExpiryThunk.fulfilled.match(action)) {
        setShowEditExpiry(false);
        await dispatch(fetchAdmins());
      } else {
        alert(action.payload || 'Failed to update expiry');
      }
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to update expiry');
    } finally {
      setLoading(false);
    }
  };

  const deleteAdmin = async () => {
    if (!selected) return;
    try {
      setLoading(true);
      const action = await dispatch(deleteAdminThunk(selected.id));
      if (deleteAdminThunk.fulfilled.match(action)) {
        setShowDelete(false);
      } else {
        alert(action.payload || 'Failed to delete admin');
      }
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to delete admin');
    } finally {
      setLoading(false);
    }
  };

  const applyAssignCats = async (e) => {
    e.preventDefault();
    if (!selected) return;
    try {
      setLoading(true);
      const picked = Array.from(assignCatsRef.current.selectedOptions).map((o) => o.value);
      const mode = assignModeRef.current.value;
      const action = await dispatch(assignAdminCategories({ id: selected.id, categoryIds: picked, mode }));
      if (assignAdminCategories.fulfilled.match(action)) {
        setShowAssignCats(false);
        setSelected(null);
      } else {
        alert(action.payload || 'Failed to update categories');
      }
    } catch (e) {
      alert(e?.response?.data?.message || 'Failed to update categories');
    } finally {
      setLoading(false);
    }
  };

  // Derived pagination
  const pageCount = Math.max(1, Math.ceil(filtered.length / pageSize));
  const safePage = Math.min(page, pageCount);
  const startIdx = (safePage - 1) * pageSize;
  const shown = filtered.slice(startIdx, startIdx + pageSize);

  const expirySeconds = (expiry) => {
    if (!expiry) return null;
    const ts = Date.parse(expiry);
    if (Number.isNaN(ts)) return null;
    const secs = Math.floor((ts - Date.now()) / 1000);
    return secs > 0 ? secs : 0;
  };

  return (
    <div className="px-3 sm:px-4 py-3">
      <div className="rounded-md border bg-white shadow-sm">
        {/* Toolbar */}
        <div className="flex flex-wrap items-center justify-between gap-2 border-b p-3">
          <h2 className="text-base font-semibold text-gray-800">Admin Management</h2>
          <div className="flex flex-wrap items-center gap-2">
            <div className="relative w-[280px] max-w-full">
              <MagnifyingGlassIcon className="pointer-events-none absolute left-2 top-1/2 h-4 w-4 -translate-y-1/2 text-blue-600" />
              <input
                type="search"
                placeholder="Search admins..."
                value={query}
                onChange={(e) => { setQuery(e.target.value); setPage(1); }}
                className="w-full rounded-md border border-gray-300 pl-7 pr-8 py-1.5 text-sm shadow-sm focus:border-blue-500 focus:outline-none focus:ring-1 focus:ring-blue-500"
              />
              {query && (
                <button onClick={() => { setQuery(''); setPage(1); }} title="Clear" className="absolute right-1 top-1/2 -translate-y-1/2 rounded p-1 text-gray-500 hover:bg-gray-100">
                  <XCircleIcon className="h-4 w-4" />
                </button>
              )}
            </div>
            <label className="hidden sm:inline-flex items-center gap-2 text-sm text-gray-700">
              <Switch
                checked={dense}
                onChange={setDense}
                className={clsx(dense ? 'bg-blue-600' : 'bg-gray-200', 'relative inline-flex h-5 w-10 items-center rounded-full transition-colors')}
              >
                <span className={clsx(dense ? 'translate-x-5' : 'translate-x-1', 'inline-block h-4 w-4 transform rounded-full bg-white transition')} />
              </Switch>
              Compact
            </label>
            <button onClick={loadAdmins} disabled={adminsLoading || loading} title="Refresh" className="inline-flex items-center gap-1 rounded-md border border-gray-300 bg-white px-2 py-1 text-sm text-gray-700 shadow-sm hover:bg-gray-50 disabled:opacity-50">
              <ArrowPathIcon className={clsx('h-4 w-4', (adminsLoading || loading) && 'animate-spin')} />
              <span className="hidden sm:inline">Refresh</span>
            </button>
            <button onClick={startCreate} disabled={loading} className="inline-flex items-center gap-1 rounded-md bg-blue-600 px-2.5 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700 disabled:opacity-50 btn btn-primary">
              <PlusIcon className="h-4 w-4" />
              New Admin
            </button>
          </div>
        </div>

        {/* Error banner */}
        {(error || adminsError) && (
          <div className="mx-3 my-2 rounded-md border border-red-200 bg-red-50 p-2 text-sm text-red-700">
            {error || adminsError}
          </div>
        )}

        {/* Table */}
        <div className="table-responsive rounded-3 border">
          <table className={clsx('table table-bordered table-hover align-middle mb-0', dense && 'table-sm')}>
            <thead className="table-light sticky-top">
              <tr>
                <th onClick={() => setSortKey('username')} className="cursor-pointer text-start">
                  <span className="inline-flex items-center gap-1">Username {sort.key==='username' && (sort.dir==='asc' ? <ChevronUpIcon className="h-4 w-4"/> : <ChevronDownIcon className="h-4 w-4"/> )}</span>
                </th>
                <th onClick={() => setSortKey('expiry')} className="cursor-pointer text-start">
                  <span className="inline-flex items-center gap-1">Expiry {sort.key==='expiry' && (sort.dir==='asc' ? <ChevronUpIcon className="h-4 w-4"/> : <ChevronDownIcon className="h-4 w-4"/> )}</span>
                </th>
                <th className="text-start">Expires In</th>
                <th className="text-start">ID</th>
                <th className="text-end">Actions</th>
              </tr>
            </thead>
            <tbody>
              {adminsLoading ? (
                <tr>
                  <td colSpan="5" className="text-center py-4 text-gray-500">
                    <ArrowPathIcon className="mx-auto h-5 w-5 animate-spin text-blue-600" />
                  </td>
                </tr>
              ) : shown.length === 0 ? (
                <tr>
                  <td colSpan="5" className="py-5">
                    <div className="text-center text-gray-500">
                      <TagIcon className="mx-auto mb-2 h-10 w-10" />
                      <div className="mb-3">No admins found</div>
                      <button onClick={startCreate} className="inline-flex items-center gap-1 rounded-md bg-blue-600 px-3 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700">
                        <PlusIcon className="h-4 w-4" />
                        Create Admin
                      </button>
                    </div>
                  </td>
                </tr>
              ) : (
                shown.map((a) => {
                  const secs = expirySeconds(a.expiry);
                  return (
                    <tr key={a.id}>
                      <td><span className="font-medium text-gray-800">{highlight(a.username, query)}</span></td>
                      <td className="text-gray-700">{highlight(formatDateTime(a.expiry), query)}</td>
                      <td>
                        {secs == null ? (
                          <span className="badge bg-secondary-subtle text-secondary-emphasis">N/A</span>
                        ) : (
                          <span className={clsx('badge', secs === 0 ? 'bg-danger-subtle text-danger-emphasis' : 'bg-primary-subtle text-primary-emphasis')}>
                            {formatHMS(secs)}
                          </span>
                        )}
                      </td>
                      <td className="font-mono text-xs text-red-700">{a.id}</td>
                      <td className="text-end">
                        <Menu as="div" className="relative inline-block text-left">
                          <Menu.Button
                            className="inline-flex items-center rounded-md border border-gray-300 bg-white p-1 text-gray-700 shadow-sm hover:bg-gray-50"
                            onClick={(e) => {
                              const rect = e.currentTarget.getBoundingClientRect();
                              const spaceBelow = window.innerHeight - rect.bottom;
                              const spaceAbove = rect.top;
                              const estHeight = 240; // approx. menu height
                              const dir = spaceBelow < estHeight && spaceAbove > spaceBelow ? 'up' : 'down';
                              setMenuDir({ id: a.id, dir });
                            }}
                          >
                            <EllipsisVerticalIcon className="h-5 w-5" />
                            <span className="sr-only">Actions</span>
                          </Menu.Button>
                          <Transition as={Fragment} enter="transition ease-out duration-100" enterFrom="transform opacity-0 scale-95" enterTo="transform opacity-100 scale-100" leave="transition ease-in duration-75" leaveFrom="transform opacity-100 scale-100" leaveTo="opacity-0 scale-95">
                            <Menu.Items
                              className={clsx(
                                'absolute right-0 z-[2000] w-60 rounded-md bg-white p-1 shadow-lg ring-1 ring-black/5 focus:outline-none',
                                menuDir.id === a.id && menuDir.dir === 'up'
                                  ? 'bottom-full mb-2 origin-bottom-right'
                                  : 'mt-2 origin-top-right'
                              )}
                            >
                              <Menu.Item>
                                {({ active }) => (
                                  <button onClick={() => startAssignCats(a)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-blue-50 text-blue-700' : 'text-gray-700')}>
                                    <TagIcon className="h-5 w-5" /> Assign categories
                                  </button>
                                )}
                              </Menu.Item>
                              <Menu.Item>
                                {({ active }) => (
                                  <button onClick={() => startDetails(a)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-blue-50 text-blue-700' : 'text-gray-700')}>
                                    <EyeIcon className="h-5 w-5" /> View details
                                  </button>
                                )}
                              </Menu.Item>
                              <Menu.Item>
                                {({ active }) => (
                                  <button onClick={() => startEditPwd(a)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-blue-50 text-blue-700' : 'text-gray-700')}>
                                    <KeyIcon className="h-5 w-5" /> Change password
                                  </button>
                                )}
                              </Menu.Item>
                              <Menu.Item>
                                {({ active }) => (
                                  <button onClick={() => startEditExpiry(a)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-blue-50 text-blue-700' : 'text-gray-700')}>
                                    <CalendarDaysIcon className="h-5 w-5" /> Extend expiry
                                  </button>
                                )}
                              </Menu.Item>
                              <div className="my-1 h-px bg-gray-100" />
                              <Menu.Item>
                                {({ active }) => (
                                  <button onClick={() => startDelete(a)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-red-50 text-red-700' : 'text-red-600')}>
                                    <TrashIcon className="h-5 w-5" /> Delete
                                  </button>
                                )}
                              </Menu.Item>
                            </Menu.Items>
                          </Transition>
                        </Menu>
                      </td>
                    </tr>
                  );
                })
              )}
            </tbody>
          </table>
        </div>

        {/* Footer / Pagination */}
        <div className="flex flex-wrap items-center justify-between gap-2 border-t p-3">
          <small className="text-gray-600">Showing {shown.length} of {filtered.length} result(s)</small>
          <div className="flex items-center gap-2">
            <select className="rounded-md border border-gray-300 bg-white px-2 py-1 text-sm shadow-sm focus:border-blue-500 focus:outline-none" value={pageSize} onChange={(e) => { setPageSize(parseInt(e.target.value, 10)); setPage(1); }}>
              {[5,10,20,50].map(n => <option key={n} value={n}>{n} / page</option>)}
            </select>
            <div className="inline-flex items-center">
              <button className="rounded-l-md border border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage<=1} onClick={() => setPage(1)} title="First"><ChevronDoubleLeftIcon className="h-5 w-5"/></button>
              <button className="border-y border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage<=1} onClick={() => setPage(safePage-1)} title="Prev"><ChevronLeftIcon className="h-5 w-5"/></button>
              <span className="border-y border-gray-300 bg-gray-50 px-2 py-1 text-sm text-gray-700">{safePage} / {pageCount}</span>
              <button className="border-y border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage>=pageCount} onClick={() => setPage(safePage+1)} title="Next"><ChevronRightIcon className="h-5 w-5"/></button>
              <button className="rounded-r-md border border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage>=pageCount} onClick={() => setPage(pageCount)} title="Last"><ChevronDoubleRightIcon className="h-5 w-5"/></button>
            </div>
          </div>
        </div>
      </div>

      {/* Create Admin Dialog */}
      <Transition show={showCreate} as={Fragment}>
        <Dialog as="div" className="relative z-50" onClose={() => setShowCreate(false)}>
          <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0" enterTo="opacity-100" leave="ease-in duration-150" leaveFrom="opacity-100" leaveTo="opacity-0">
            <div className="fixed inset-0 bg-black/30" />
          </Transition.Child>
          <div className="fixed inset-0 overflow-y-auto">
            <div className="flex min-h-full items-center justify-center p-4">
              <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0 scale-95" enterTo="opacity-100 scale-100" leave="ease-in duration-150" leaveFrom="opacity-100 scale-100" leaveTo="opacity-0 scale-95">
                <Dialog.Panel className="w-full max-w-md rounded-lg bg-white p-4 shadow-xl">
                  <Dialog.Title className="text-base font-semibold text-gray-800">Create Admin</Dialog.Title>
                  <form className="mt-3 space-y-3" onSubmit={createAdmin}>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">Username</label>
                      <input ref={usernameRef} required className="w-full rounded-md border border-gray-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none" />
                    </div>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">Password</label>
                      <input ref={passwordRef} type="password" required className="w-full rounded-md border border-gray-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none" />
                    </div>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">App Category</label>
                      <select ref={createCatRef} required defaultValue="" className="w-full rounded-md border border-gray-300 bg-white px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none">
                        <option value="" disabled>-- Select a category --</option>
                        {categoriesState.items.map((c) => (
                          <option key={c.id} value={c.id}>{c.title}</option>
                        ))}
                      </select>
                      <div className="mt-1 text-xs text-gray-500">Value is the id; shown text is the category name.</div>
                    </div>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">Expiry (days)</label>
                      <input ref={expiryRef} type="number" min={1} defaultValue={30} className="w-full rounded-md border border-gray-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none" />
                    </div>
                    <div className="mt-4 flex justify-end gap-2">
                      <button type="button" onClick={() => setShowCreate(false)} className="rounded-md border border-gray-300 bg-white px-3 py-1.5 text-sm text-gray-700 shadow-sm hover:bg-gray-50">Cancel</button>
                      <button type="submit" disabled={loading} className="rounded-md bg-blue-600 px-3 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700 disabled:opacity-50">Create</button>
                    </div>
                  </form>
                </Dialog.Panel>
              </Transition.Child>
            </div>
          </div>
        </Dialog>
      </Transition>

      {/* Edit Password Dialog */}
      <Transition show={showEditPwd} as={Fragment}>
        <Dialog as="div" className="relative z-50" onClose={() => setShowEditPwd(false)}>
          <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0" enterTo="opacity-100" leave="ease-in duration-150" leaveFrom="opacity-100" leaveTo="opacity-0">
            <div className="fixed inset-0 bg-black/30" />
          </Transition.Child>
          <div className="fixed inset-0 overflow-y-auto">
            <div className="flex min-h-full items-center justify-center p-4">
              <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0 scale-95" enterTo="opacity-100 scale-100" leave="ease-in duration-150" leaveFrom="opacity-100 scale-100" leaveTo="opacity-0 scale-95">
                <Dialog.Panel className="w-full max-w-md rounded-lg bg-white p-4 shadow-xl">
                  <Dialog.Title className="text-base font-semibold text-gray-800">Change Password</Dialog.Title>
                  <form className="mt-3 space-y-3" onSubmit={updatePassword}>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">New password for {selected?.username}</label>
                      <input ref={newPwdRef} type="password" required className="w-full rounded-md border border-gray-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none" />
                    </div>
                    <div className="mt-4 flex justify-end gap-2">
                      <button type="button" onClick={() => setShowEditPwd(false)} className="rounded-md border border-gray-300 bg-white px-3 py-1.5 text-sm text-gray-700 shadow-sm hover:bg-gray-50">Cancel</button>
                      <button type="submit" disabled={loading} className="rounded-md bg-blue-600 px-3 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700 disabled:opacity-50">Update</button>
                    </div>
                  </form>
                </Dialog.Panel>
              </Transition.Child>
            </div>
          </div>
        </Dialog>
      </Transition>

      {/* Edit Expiry Dialog */}
      <Transition show={showEditExpiry} as={Fragment}>
        <Dialog as="div" className="relative z-50" onClose={() => setShowEditExpiry(false)}>
          <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0" enterTo="opacity-100" leave="ease-in duration-150" leaveFrom="opacity-100" leaveTo="opacity-0">
            <div className="fixed inset-0 bg-black/30" />
          </Transition.Child>
          <div className="fixed inset-0 overflow-y-auto">
            <div className="flex min-h-full items-center justify-center p-4">
              <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0 scale-95" enterTo="opacity-100 scale-100" leave="ease-in duration-150" leaveFrom="opacity-100 scale-100" leaveTo="opacity-0 scale-95">
                <Dialog.Panel className="w-full max-w-md rounded-lg bg-white p-4 shadow-xl">
                  <Dialog.Title className="text-base font-semibold text-gray-800">Update Expiry</Dialog.Title>
                  <form className="mt-3 space-y-3" onSubmit={updateExpiry}>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">Extend days for {selected?.username}</label>
                      <input ref={newExpiryRef} type="number" min={1} required className="w-full rounded-md border border-gray-300 px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none" />
                    </div>
                    <div className="mt-4 flex justify-end gap-2">
                      <button type="button" onClick={() => setShowEditExpiry(false)} className="rounded-md border border-gray-300 bg-white px-3 py-1.5 text-sm text-gray-700 shadow-sm hover:bg-gray-50">Cancel</button>
                      <button type="submit" disabled={loading} className="rounded-md bg-blue-600 px-3 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700 disabled:opacity-50">Update</button>
                    </div>
                  </form>
                </Dialog.Panel>
              </Transition.Child>
            </div>
          </div>
        </Dialog>
      </Transition>

      {/* Delete Confirm Dialog */}
      <Transition show={showDelete} as={Fragment}>
        <Dialog as="div" className="relative z-50" onClose={() => setShowDelete(false)}>
          <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0" enterTo="opacity-100" leave="ease-in duration-150" leaveFrom="opacity-100" leaveTo="opacity-0">
            <div className="fixed inset-0 bg-black/30" />
          </Transition.Child>
          <div className="fixed inset-0 overflow-y-auto">
            <div className="flex min-h-full items-center justify-center p-4">
              <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0 scale-95" enterTo="opacity-100 scale-100" leave="ease-in duration-150" leaveFrom="opacity-100 scale-100" leaveTo="opacity-0 scale-95">
                <Dialog.Panel className="w-full max-w-md rounded-lg bg-white p-4 shadow-xl">
                  <Dialog.Title className="text-base font-semibold text-gray-800">Delete Admin</Dialog.Title>
                  <div className="mt-3 text-sm text-gray-700">Are you sure you want to delete <strong className="font-semibold">{selected?.username}</strong>? This action cannot be undone.</div>
                  <div className="mt-4 flex justify-end gap-2">
                    <button type="button" onClick={() => setShowDelete(false)} className="rounded-md border border-gray-300 bg-white px-3 py-1.5 text-sm text-gray-700 shadow-sm hover:bg-gray-50">Cancel</button>
                    <button type="button" onClick={deleteAdmin} disabled={loading} className="rounded-md bg-red-600 px-3 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-red-700 disabled:opacity-50">Delete</button>
                  </div>
                </Dialog.Panel>
              </Transition.Child>
            </div>
          </div>
        </Dialog>
      </Transition>

      {/* Details Dialog */}
      <Transition show={showDetails} as={Fragment}>
        <Dialog as="div" className="relative z-50" onClose={() => setShowDetails(false)}>
          <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0" enterTo="opacity-100" leave="ease-in duration-150" leaveFrom="opacity-100" leaveTo="opacity-0">
            <div className="fixed inset-0 bg-black/30" />
          </Transition.Child>
          <div className="fixed inset-0 overflow-y-auto">
            <div className="flex min-h-full items-center justify-center p-4">
              <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0 scale-95" enterTo="opacity-100 scale-100" leave="ease-in duration-150" leaveFrom="opacity-100 scale-100" leaveTo="opacity-0 scale-95">
                <Dialog.Panel className="w-full max-w-lg rounded-lg bg-white p-4 shadow-xl">
                  <Dialog.Title className="text-base font-semibold text-gray-800">Admin Details</Dialog.Title>
                  <div className="mt-3 text-sm text-gray-700">
                    <div className="grid grid-cols-12 gap-x-3 gap-y-2">
                      <div className="col-span-4 text-gray-500">Username</div>
                      <div className="col-span-8 font-medium">{selected?.username}</div>
                      <div className="col-span-4 text-gray-500">ID</div>
                      <div className="col-span-8 font-mono text-xs">{selected?.id}</div>
                      <div className="col-span-4 text-gray-500">Expiry</div>
                      <div className="col-span-8">{selected?.expiry || 'N/A'}</div>
                      <div className="col-span-4 text-gray-500">Expires In</div>
                      <div className="col-span-8">{formatHMS(expirySeconds(selected?.expiry) ?? 0)}</div>
                      <div className="col-span-4 text-gray-500">Categories</div>
                      <div className="col-span-8">
                        {detailsCategories.length === 0 ? (
                          <span className="text-gray-500">None</span>
                        ) : (
                          <div className="flex flex-wrap gap-1">
                            {detailsCategories.map((c) => (
                              <span key={c.id} className="inline-flex items-center rounded-full bg-blue-50 px-2 py-0.5 text-xs font-medium text-blue-700 ring-1 ring-blue-200">{c.title}</span>
                            ))}
                          </div>
                        )}
                      </div>
                    </div>
                  </div>
                  <div className="mt-4 flex justify-end">
                    <button type="button" onClick={() => setShowDetails(false)} className="rounded-md border border-gray-300 bg-white px-3 py-1.5 text-sm text-gray-700 shadow-sm hover:bg-gray-50">Close</button>
                  </div>
                </Dialog.Panel>
              </Transition.Child>
            </div>
          </div>
        </Dialog>
      </Transition>

      {/* Assign Categories Dialog */}
      <Transition show={showAssignCats && !!selected} as={Fragment}>
        <Dialog as="div" className="relative z-50" onClose={() => setShowAssignCats(false)}>
          <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0" enterTo="opacity-100" leave="ease-in duration-150" leaveFrom="opacity-100" leaveTo="opacity-0">
            <div className="fixed inset-0 bg-black/30" />
          </Transition.Child>
          <div className="fixed inset-0 overflow-y-auto">
            <div className="flex min-h-full items-center justify-center p-4">
              <Transition.Child as={Fragment} enter="ease-out duration-200" enterFrom="opacity-0 scale-95" enterTo="opacity-100 scale-100" leave="ease-in duration-150" leaveFrom="opacity-100 scale-100" leaveTo="opacity-0 scale-95">
                <Dialog.Panel className="w-full max-w-lg rounded-lg bg-white p-4 shadow-xl">
                  <Dialog.Title className="text-base font-semibold text-gray-800">Assign Categories to {selected?.username}</Dialog.Title>
                  <form className="mt-3 space-y-3" onSubmit={applyAssignCats}>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">Mode</label>
                      <select ref={assignModeRef} className="w-full rounded-md border border-gray-300 bg-white px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none">
                        <option value="add">Add selected</option>
                        <option value="remove">Remove selected</option>
                        <option value="replace">Replace (set exactly)</option>
                      </select>
                    </div>
                    <div>
                      <label className="mb-1 block text-sm font-medium text-gray-700">Categories</label>
                      <select ref={assignCatsRef} multiple size={8} defaultValue={assignedCategoryIds} className="w-full rounded-md border border-gray-300 bg-white px-3 py-2 text-sm shadow-sm focus:border-blue-500 focus:outline-none">
                        {categoriesState.items.map((c) => (
                          <option key={c.id} value={c.id}>{c.title}</option>
                        ))}
                      </select>
                      <div className="mt-1 text-xs text-gray-500">Hold Ctrl/Cmd to multi-select. Already assigned are preselected.</div>
                    </div>
                    <div className="mt-4 flex justify-end gap-2">
                      <button type="button" onClick={() => setShowAssignCats(false)} className="rounded-md border border-gray-300 bg-white px-3 py-1.5 text-sm text-gray-700 shadow-sm hover:bg-gray-50">Cancel</button>
                      <button type="submit" disabled={loading} className="rounded-md bg-blue-600 px-3 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700 disabled:opacity-50">Apply</button>
                    </div>
                  </form>
                </Dialog.Panel>
              </Transition.Child>
            </div>
          </div>
        </Dialog>
      </Transition>
    </div>
  );
};

export default OwnerAdmins;

import React, { Fragment, useEffect, useMemo, useRef, useState } from 'react';
import { useDispatch, useSelector } from 'react-redux';
import {
  fetchCategories as fetchCategoriesThunk,
  createCategory as createCategoryThunk,
  updateCategory as updateCategoryThunk,
  deleteCategory as deleteCategoryThunk,
  toggleCategoryStatus as toggleCategoryStatusThunk,
  assignCategoryAdmins as assignCategoryAdminsThunk,
} from '../../features/categories/categoriesSlice.js';
import { fetchAdmins as fetchAdminsThunk } from '../../features/admins/adminsSlice.js';
import { Dialog, Menu, Switch, Transition } from '@headlessui/react';
import {
  AdjustmentsHorizontalIcon,
  ArrowPathIcon,
  CheckCircleIcon,
  ChevronDoubleLeftIcon,
  ChevronDoubleRightIcon,
  ChevronDownIcon,
  ChevronLeftIcon,
  ChevronRightIcon,
  ChevronUpIcon,
  EllipsisVerticalIcon,
  LinkIcon,
  MagnifyingGlassIcon,
  PencilSquareIcon,
  PlusIcon,
  TagIcon,
  TrashIcon,
  UserPlusIcon,
  XCircleIcon,
} from '@heroicons/react/24/outline';

function clsx(...classes) {
  return classes.filter(Boolean).join(' ');
}

const OwnerCategories = () => {
  const dispatch = useDispatch();
  const categoriesState = useSelector((s) => s.categories);
  const adminsState = useSelector((s) => s.admins);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [query, setQuery] = useState('');
  const categories = categoriesState.items;
  const stats = categoriesState.stats;
  const [includeInactive, setIncludeInactive] = useState(true);
  const [dense, setDense] = useState(true);

  const [sort, setSort] = useState({ key: 'title', dir: 'asc' });

  // Admins for assignment
  const admins = adminsState.items;

  // Pagination
  const [page, setPage] = useState(1);
  const [pageSize, setPageSize] = useState(10);

  // Modals
  const [showCreate, setShowCreate] = useState(false);
  const [showEdit, setShowEdit] = useState(false);
  const [showDelete, setShowDelete] = useState(false);
  const [showAssign, setShowAssign] = useState(false);
  const [selected, setSelected] = useState(null);
  // Dropdown direction state (flip up when near bottom)
  const [menuDir, setMenuDir] = useState({ id: null, dir: 'down' });

  // Form refs
  const titleRef = useRef(null);
  const blogRef = useRef(null);
  const linkRef = useRef(null);
  const editTitleRef = useRef(null);
  const editBlogRef = useRef(null);
  const editLinkRef = useRef(null);
  const assignSelectRef = useRef(null);
  const assignModeRef = useRef(null);

  const load = async () => {
    setLoading(true);
    setError('');
    try {
      await Promise.all([
        dispatch(fetchCategoriesThunk({ includeInactive })),
        dispatch(fetchAdminsThunk()),
      ]);
    } catch (err) {
      setError(err?.response?.data?.message || 'Failed to load categories');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    load();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [includeInactive]);

  const setSortKey = (key) => {
    setSort((s) => ({ key, dir: s.key === key ? (s.dir === 'asc' ? 'desc' : 'asc') : 'asc' }));
  };

  const highlight = (text, q) => {
    const t = (text ?? '').toString();
    const s = q.trim();
    if (!s) return t;
    try {
      const idx = t.toLowerCase().indexOf(s.toLowerCase());
      if (idx === -1) return t;
      const before = t.slice(0, idx);
      const match = t.slice(idx, idx + s.length);
      const after = t.slice(idx + s.length);
      return (<span>{before}<mark className="px-0 py-0 bg-warning-subtle text-dark" style={{borderRadius:2}}>{match}</mark>{after}</span>);
    } catch {
      return t;
    }
  };

  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    if (!q) return categories;
    return categories.filter(c =>
      c.title?.toLowerCase().includes(q) ||
      c.blog?.toLowerCase().includes(q) ||
      c.downloadLink?.toLowerCase().includes(q)
    );
  }, [categories, query]);

  const sorted = useMemo(() => {
    const arr = [...filtered];
    const { key, dir } = sort;
    const fact = dir === 'asc' ? 1 : -1;
    arr.sort((a, b) => {
      let va, vb;
      if (key === 'status') { va = a.active ? 1 : 0; vb = b.active ? 1 : 0; }
      else if (key === 'title') { va = a.title || ''; vb = b.title || ''; }
      else if (key === 'blog') { va = a.blog || ''; vb = b.blog || ''; }
      else if (key === 'link') { va = a.downloadLink || ''; vb = b.downloadLink || ''; }
      else { va = a.title || ''; vb = b.title || ''; }
      if (typeof va === 'string') va = va.toLowerCase();
      if (typeof vb === 'string') vb = vb.toLowerCase();
      if (va < vb) return -1 * fact;
      if (va > vb) return 1 * fact;
      return 0;
    });
    return arr;
  }, [filtered, sort]);

  const pageCount = Math.max(1, Math.ceil(sorted.length / pageSize));
  const safePage = Math.min(page, pageCount);
  const startIdx = (safePage - 1) * pageSize;
  const shown = sorted.slice(startIdx, startIdx + pageSize);

  const startCreate = () => {
    setShowCreate(true);
    setTimeout(() => titleRef.current?.focus(), 100);
  };
  const startEdit = (cat) => {
    setSelected(cat);
    setShowEdit(true);
    setTimeout(() => editTitleRef.current?.focus(), 100);
  };
  const startDelete = (cat) => {
    setSelected(cat);
    setShowDelete(true);
  };
  const startAssign = (cat) => {
    setSelected(cat);
    setShowAssign(true);
    setTimeout(() => assignSelectRef.current?.focus(), 100);
  };

  const createCategory = async (e) => {
    e.preventDefault();
    try {
      setLoading(true);
      const action = await dispatch(createCategoryThunk({
        title: titleRef.current.value,
        blog: blogRef.current.value,
        downloadLink: linkRef.current.value,
      }));
      if (!createCategoryThunk.fulfilled.match(action)) throw new Error(action.payload || 'Failed');
      setShowCreate(false);
      titleRef.current.value = '';
      blogRef.current.value = '';
      linkRef.current.value = '';
      // state already updated by slice
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to create category');
    } finally {
      setLoading(false);
    }
  };

  const updateCategory = async (e) => {
    e.preventDefault();
    if (!selected) return;
    try {
      setLoading(true);
      const action = await dispatch(updateCategoryThunk({ id: selected.id, patch: {
        title: editTitleRef.current.value,
        blog: editBlogRef.current.value,
        downloadLink: editLinkRef.current.value,
      }}));
      if (!updateCategoryThunk.fulfilled.match(action)) throw new Error(action.payload || 'Failed');
      setShowEdit(false);
      setSelected(null);
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to update category');
    } finally {
      setLoading(false);
    }
  };

  const deleteCategory = async () => {
    if (!selected) return;
    try {
      setLoading(true);
      const action = await dispatch(deleteCategoryThunk(selected.id));
      if (!deleteCategoryThunk.fulfilled.match(action)) throw new Error(action.payload || 'Failed');
      setShowDelete(false);
      setSelected(null);
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to delete category');
    } finally {
      setLoading(false);
    }
  };

  const toggleActive = async (cat) => {
    try {
      setLoading(true);
      const action = await dispatch(toggleCategoryStatusThunk({ id: cat.id, active: !cat.active }));
      if (!toggleCategoryStatusThunk.fulfilled.match(action)) throw new Error(action.payload || 'Failed');
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to update status');
    } finally {
      setLoading(false);
    }
  };

  const assignToAdmins = async (e) => {
    e.preventDefault();
    if (!selected) return;
    try {
      setLoading(true);
      const options = Array.from(assignSelectRef.current.selectedOptions).map(o => o.value);
      const mode = assignModeRef.current.value;
      const action = await dispatch(assignCategoryAdminsThunk({ id: selected.id, adminIds: options, mode }));
      if (!assignCategoryAdminsThunk.fulfilled.match(action)) throw new Error(action.payload || 'Failed');
      setShowAssign(false);
      setSelected(null);
    } catch (err) {
      alert(err?.response?.data?.message || 'Failed to assign category');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="px-3 sm:px-4 py-3">
      <div className="rounded-md border bg-white shadow-sm">
        {/* Toolbar */}
        <div className="flex flex-wrap items-center justify-between gap-2 border-b p-3">
          <h2 className="text-base font-semibold text-gray-800">App Categories</h2>
          <div className="flex flex-wrap items-center gap-2">
            {/* Search */}
            <div className="relative w-[320px] max-w-full">
              <MagnifyingGlassIcon className="pointer-events-none absolute left-2 top-1/2 h-4 w-4 -translate-y-1/2 text-blue-600" />
              <input
                type="search"
                value={query}
                onChange={(e) => { setQuery(e.target.value); setPage(1); }}
                placeholder="Search categories..."
                className="w-full rounded-md border border-gray-300 pl-7 pr-8 py-1.5 text-sm shadow-sm focus:border-blue-500 focus:outline-none focus:ring-1 focus:ring-blue-500"
              />
              {query && (
                <button
                  onClick={() => { setQuery(''); setPage(1); }}
                  title="Clear"
                  className="absolute right-1 top-1/2 -translate-y-1/2 rounded p-1 text-gray-500 hover:bg-gray-100"
                >
                  <XCircleIcon className="h-4 w-4" />
                </button>
              )}
            </div>
            {/* Toggles */}
            <div className="hidden sm:flex items-center gap-3">
              <label className="flex items-center gap-2 text-sm text-gray-700">
                <Switch
                  checked={includeInactive}
                  onChange={setIncludeInactive}
                  className={clsx(
                    includeInactive ? 'bg-blue-600' : 'bg-gray-200',
                    'relative inline-flex h-5 w-10 items-center rounded-full transition-colors'
                  )}
                >
                  <span
                    className={clsx(
                      includeInactive ? 'translate-x-5' : 'translate-x-1',
                      'inline-block h-4 w-4 transform rounded-full bg-white transition'
                    )}
                  />
                </Switch>
                Include inactive
              </label>
              <label className="flex items-center gap-2 text-sm text-gray-700">
                <Switch
                  checked={dense}
                  onChange={setDense}
                  className={clsx(
                    dense ? 'bg-blue-600' : 'bg-gray-200',
                    'relative inline-flex h-5 w-10 items-center rounded-full transition-colors'
                  )}
                >
                  <span
                    className={clsx(
                      dense ? 'translate-x-5' : 'translate-x-1',
                      'inline-block h-4 w-4 transform rounded-full bg-white transition'
                    )}
                  />
                </Switch>
                Compact
              </label>
            </div>
            <button
              onClick={load}
              disabled={loading}
              title="Refresh"
              className="inline-flex items-center gap-1 rounded-md border border-gray-300 bg-white px-2 py-1 text-sm text-gray-700 shadow-sm hover:bg-gray-50 disabled:opacity-50"
            >
              <ArrowPathIcon className={clsx('h-4 w-4', loading && 'animate-spin')} />
              <span className="hidden sm:inline">Refresh</span>
            </button>
            <button
              onClick={startCreate}
              disabled={loading}
              className="inline-flex items-center gap-1 rounded-md bg-blue-600 px-2.5 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700 disabled:opacity-50"
            >
              <PlusIcon className="h-4 w-4" />
              New Category
            </button>
          </div>
        </div>

        {/* Stats and error */}
        {error && (
          <div className="mx-3 mt-3 rounded-md border border-red-200 bg-red-50 p-3 text-sm text-red-700">{error}</div>
        )}
        {stats && (
          <div className="flex flex-wrap items-center gap-2 px-3 pt-3">
            <span className="inline-flex items-center rounded-full bg-blue-50 px-2 py-1 text-xs font-medium text-blue-700">Total: {stats.totalCategories}</span>
            <span className="inline-flex items-center rounded-full bg-green-50 px-2 py-1 text-xs font-medium text-green-700">Active: {stats.activeCategories}</span>
            <span className="inline-flex items-center rounded-full bg-gray-100 px-2 py-1 text-xs font-medium text-gray-700">Inactive: {stats.inactiveCategories}</span>
          </div>
        )}

        {/* Table */}
        <div className="mt-3 table-responsive rounded-3 border">
          <table className={clsx('table table-bordered table-hover align-middle mb-0', dense && 'table-sm')}>
            <thead className="table-light sticky-top">
              <tr>
                <th onClick={() => setSortKey('title')} className="cursor-pointer text-start">
                  <span className="inline-flex items-center gap-1">Title {sort.key==='title' && (sort.dir==='asc' ? <ChevronUpIcon className="h-4 w-4"/> : <ChevronDownIcon className="h-4 w-4"/> )}</span>
                </th>
                <th onClick={() => setSortKey('blog')} className="cursor-pointer text-start">
                  <span className="inline-flex items-center gap-1">Blog {sort.key==='blog' && (sort.dir==='asc' ? <ChevronUpIcon className="h-4 w-4"/> : <ChevronDownIcon className="h-4 w-4"/> )}</span>
                </th>
                <th onClick={() => setSortKey('link')} className="cursor-pointer text-start">
                  <span className="inline-flex items-center gap-1">Download Link {sort.key==='link' && (sort.dir==='asc' ? <ChevronUpIcon className="h-4 w-4"/> : <ChevronDownIcon className="h-4 w-4"/> )}</span>
                </th>
                <th onClick={() => setSortKey('status')} className="cursor-pointer text-start">
                  <span className="inline-flex items-center gap-1">Status {sort.key==='status' && (sort.dir==='asc' ? <ChevronUpIcon className="h-4 w-4"/> : <ChevronDownIcon className="h-4 w-4"/> )}</span>
                </th>
                <th className="text-end">Actions</th>
              </tr>
            </thead>
            <tbody>
              {loading ? (
                <tr>
                  <td colSpan="5" className="text-center py-4 text-gray-500">
                    <ArrowPathIcon className="mx-auto h-5 w-5 animate-spin text-blue-600" />
                  </td>
                </tr>
              ) : shown.length === 0 ? (
                <tr>
                  <td colSpan="5" className="py-5">
                    <div className="text-center text-gray-500">
                      <TagIcon className="mx-auto mb-2 h-10 w-10" />
                      <div className="mb-3">No categories found</div>
                      <button onClick={startCreate} className="inline-flex items-center gap-1 rounded-md bg-blue-600 px-3 py-1.5 text-sm font-medium text-white shadow-sm hover:bg-blue-700">
                        <PlusIcon className="h-4 w-4" />
                        Create Category
                      </button>
                    </div>
                  </td>
                </tr>
              ) : (
                shown.map((c) => (
                  <tr key={c.id}>
                    <td>
                      <span className="font-medium text-gray-800">{highlight(c.title, query)}</span>
                    </td>
                    <td className="text-gray-700" title={c.blog}>
                      <span className="d-inline-block text-truncate" style={{maxWidth: 360}}>{highlight(c.blog, query)}</span>
                    </td>
                    <td className="text-gray-700">
                      <a className="inline-flex items-center gap-1 text-blue-600 hover:underline" href={c.downloadLink} target="_blank" rel="noreferrer">
                        <LinkIcon className="h-4 w-4" />
                        {highlight(c.downloadLink, query)}
                      </a>
                    </td>
                    <td>
                      <span className={clsx('badge', c.active ? 'bg-success-subtle text-success-emphasis' : 'bg-secondary-subtle text-secondary-emphasis')}>
                        {c.active ? 'Active' : 'Inactive'}
                      </span>
                    </td>
                    <td className="text-end">
                      <Menu as="div" className="relative inline-block text-left">
                        <Menu.Button
                          className="inline-flex items-center rounded-md border border-gray-300 bg-white p-1 text-gray-700 shadow-sm hover:bg-gray-50"
                          onClick={(e) => {
                            const rect = e.currentTarget.getBoundingClientRect();
                            const spaceBelow = window.innerHeight - rect.bottom;
                            const spaceAbove = rect.top;
                            const estHeight = 220; // approx. menu height
                            const dir = spaceBelow < estHeight && spaceAbove > spaceBelow ? 'up' : 'down';
                            setMenuDir({ id: c.id, dir });
                          }}
                        >
                          <EllipsisVerticalIcon className="h-5 w-5" />
                          <span className="sr-only">Actions</span>
                        </Menu.Button>
                        <Transition
                          as={Fragment}
                          enter="transition ease-out duration-100"
                          enterFrom="transform opacity-0 scale-95"
                          enterTo="transform opacity-100 scale-100"
                          leave="transition ease-in duration-75"
                          leaveFrom="transform opacity-100 scale-100"
                          leaveTo="transform opacity-0 scale-95"
                        >
                          <Menu.Items
                            className={clsx(
                              'absolute right-0 z-[2000] w-56 rounded-md bg-white p-1 shadow-lg ring-1 ring-black/5 focus:outline-none',
                              menuDir.id === c.id && menuDir.dir === 'up'
                                ? 'bottom-full mb-2 origin-bottom-right'
                                : 'mt-2 origin-top-right'
                            )}
                          >
                            <Menu.Item>
                              {({ active }) => (
                                <button onClick={() => startAssign(c)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-blue-50 text-blue-700' : 'text-gray-700')}>
                                  <UserPlusIcon className="h-5 w-5" /> Assign to Admins
                                </button>
                              )}
                            </Menu.Item>
                            <Menu.Item>
                              {({ active }) => (
                                <button onClick={() => startEdit(c)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-blue-50 text-blue-700' : 'text-gray-700')}>
                                  <PencilSquareIcon className="h-5 w-5" /> Edit
                                </button>
                              )}
                            </Menu.Item>
                            <Menu.Item>
                              {({ active }) => (
                                <button onClick={() => toggleActive(c)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-blue-50 text-blue-700' : 'text-gray-700')}>
                                  {c.active ? <XCircleIcon className="h-5 w-5" /> : <CheckCircleIcon className="h-5 w-5" />} {c.active ? 'Set Inactive' : 'Set Active'}
                                </button>
                              )}
                            </Menu.Item>
                            <div className="my-1 h-px bg-gray-100" />
                            <Menu.Item>
                              {({ active }) => (
                                <button onClick={() => startDelete(c)} className={clsx('flex w-full items-center gap-2 rounded-md px-3 py-2 text-left text-sm', active ? 'bg-red-50 text-red-700' : 'text-red-600')}>
                                  <TrashIcon className="h-5 w-5" /> Delete
                                </button>
                              )}
                            </Menu.Item>
                          </Menu.Items>
                        </Transition>
                      </Menu>
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>

        {/* Footer / Pagination */}
        <div className="flex flex-wrap items-center justify-between gap-2 border-t p-3">
          <small className="text-gray-600">Showing {shown.length} of {filtered.length} result(s)</small>
          <div className="flex items-center gap-2">
            <select
              className="rounded-md border border-gray-300 bg-white px-2 py-1 text-sm shadow-sm focus:border-blue-500 focus:outline-none"
              value={pageSize}
              onChange={(e) => { setPageSize(parseInt(e.target.value, 10)); setPage(1); }}
            >
              {[5, 10, 20, 50].map((n) => (
                <option key={n} value={n}>{n} / page</option>
              ))}
            </select>
            <div className="inline-flex items-center">
              <button className="rounded-l-md border border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage <= 1} onClick={() => setPage(1)} title="First">
                <ChevronDoubleLeftIcon className="h-5 w-5" />
              </button>
              <button className="border-y border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage <= 1} onClick={() => setPage(safePage - 1)} title="Prev">
                <ChevronLeftIcon className="h-5 w-5" />
              </button>
              <span className="border-y border-gray-300 bg-gray-50 px-2 py-1 text-sm text-gray-700">{safePage} / {pageCount}</span>
              <button className="border-y border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage >= pageCount} onClick={() => setPage(safePage + 1)} title="Next">
                <ChevronRightIcon className="h-5 w-5" />
              </button>
              <button className="rounded-r-md border border-gray-300 bg-white p-1 text-gray-700 hover:bg-gray-50 disabled:opacity-50" disabled={safePage >= pageCount} onClick={() => setPage(pageCount)} title="Last">
                <ChevronDoubleRightIcon className="h-5 w-5" />
              </button>
            </div>
          </div>

// Simple date-time formatting utilities

const pad = (n) => n.toString().padStart(2, '0');

/**
 * Format a date/time string into YYYY-MM-DD HH:MM:SS (local time)
 * Accepts ISO string, epoch ms, Date, or falsy.
 */
export const formatDateTime = (value) => {
  if (!value) return '-';
  const d = value instanceof Date ? value : new Date(value);
  if (Number.isNaN(d.getTime())) return '-';
  const y = d.getFullYear();
  const m = pad(d.getMonth() + 1);
  const day = pad(d.getDate());
  const h = pad(d.getHours());
  const min = pad(d.getMinutes());
  const s = pad(d.getSeconds());
  return `${y}-${m}-${day} ${h}:${min}:${s}`;
};

export default { formatDateTime };

import { jwtDecode } from 'jwt-decode';
import { AUTH } from './constants.js';

/**
 * JWT Utility Functions
 * Handles token validation, decoding, and management
 */

/**
 * Decode JWT token and extract payload
 * @param {string} token - JWT token to decode
 * @returns {object|null} - Decoded payload or null if invalid
 */
export const decodeToken = (token) => {
  try {
    if (!token) return null;
    return jwtDecode(token);
  } catch (error) {
    console.error('Error decoding token:', error);
    return null;
  }
};

/**
 * Check if token is expired
 * @param {string} token - JWT token to check
 * @returns {boolean} - True if expired, false otherwise
 */
export const isTokenExpired = (token) => {
  try {
    const decoded = decodeToken(token);
    if (!decoded || !decoded.exp) return true;
    
    const currentTime = Date.now() / 1000;
    return decoded.exp < currentTime;
  } catch (error) {
    console.error('Error checking token expiration:', error);
    return true;
  }
};

/**
 * Validate token structure and content
 * @param {string} token - JWT token to validate
 * @returns {boolean} - True if valid, false otherwise
 */
export const isValidToken = (token) => {
  try {
    if (!token) return false;
    
    const decoded = decodeToken(token);
    if (!decoded) return false;
    
    // Check required fields based on your JWT structure
    const hasRequiredFields = 
      decoded.sub &&           // Username (standard claim)
      decoded.username &&      // Username (custom claim)
      decoded.userId &&        // User ID (MongoDB ObjectId)
      decoded.role &&          // Role (ADMIN/OWNER)
      decoded.exp &&           // Expiration
      decoded.iat;             // Issued at
    
    if (!hasRequiredFields) return false;
    
    // Check if token is not expired
    return !isTokenExpired(token);
  } catch (error) {
    console.error('Error validating token:', error);
    return false;
  }
};

/**
 * Get user data from token
 * @param {string} token - JWT token
 * @returns {object|null} - User data or null if invalid
 */
export const getUserFromToken = (token) => {
  try {
    if (!isValidToken(token)) return null;
    
    const decoded = decodeToken(token);
    return {
      id: decoded.userId,
      username: decoded.username,
      role: decoded.role,
      subject: decoded.sub,
      issuedAt: decoded.iat,
      expiresAt: decoded.exp,
    };
  } catch (error) {
    console.error('Error extracting user from token:', error);
    return null;
  }
};

/**
 * Get user role from token
 * @param {string} token - JWT token
 * @returns {string|null} - User role or null if invalid
 */
export const getRoleFromToken = (token) => {
  try {
    const user = getUserFromToken(token);
    return user ? user.role : null;
  } catch (error) {
    console.error('Error extracting role from token:', error);
    return null;
  }
};

/**
 * Check if user has specific role
 * @param {string} token - JWT token
 * @param {string} role - Role to check (ADMIN/OWNER)
 * @returns {boolean} - True if user has role, false otherwise
 */
export const hasRole = (token, role) => {
  try {
    const userRole = getRoleFromToken(token);
    return userRole === role;
  } catch (error) {
    console.error('Error checking user role:', error);
    return false;
  }
};

/**
 * Get token from localStorage
 * @returns {string|null} - Token or null if not found
 */
export const getStoredToken = () => {
  try {
    return localStorage.getItem(AUTH.TOKEN_KEY);
  } catch (error) {
    console.error('Error getting stored token:', error);
    return null;
  }
};

/**
 * Store token in localStorage
 * @param {string} token - JWT token to store
 */
export const storeToken = (token) => {
  try {
    if (token) {
      localStorage.setItem(AUTH.TOKEN_KEY, token);
    }
  } catch (error) {
    console.error('Error storing token:', error);
  }
};

/**
 * Remove token from localStorage
 */
export const removeToken = () => {
  try {
    localStorage.removeItem(AUTH.TOKEN_KEY);
    localStorage.removeItem(AUTH.USER_KEY);
  } catch (error) {
    console.error('Error removing token:', error);
  }
};

/**
 * Get current authenticated user data
 * @returns {object|null} - User data or null if not authenticated
 */
export const getCurrentUser = () => {
  try {
    const token = getStoredToken();
    return getUserFromToken(token);
  } catch (error) {
    console.error('Error getting current user:', error);
    return null;
  }
};

/**
 * Check if user is currently authenticated
 * @returns {boolean} - True if authenticated, false otherwise
 */
export const isAuthenticated = () => {
  try {
    const token = getStoredToken();
    return isValidToken(token);
  } catch (error) {
    console.error('Error checking authentication:', error);
    return false;
  }
};

/**
 * Get dashboard route based on user role
 * @param {string} token - JWT token
 * @returns {string} - Dashboard route for the user's role
 */
export const getDashboardRoute = (token) => {
  try {
    const role = getRoleFromToken(token);
    
    switch (role) {
      case AUTH.ROLES.ADMIN:
        return '/admin/dashboard';
      case AUTH.ROLES.OWNER:
        return '/owner/dashboard';
      default:
        return '/login';
    }
  } catch (error) {
    console.error('Error getting dashboard route:', error);
    return '/login';
  }
};

/**
 * Format token expiration time for display
 * @param {string} token - JWT token
 * @returns {string} - Formatted expiration time
 */
export const getTokenExpirationTime = (token) => {
  try {
    const decoded = decodeToken(token);
    if (!decoded || !decoded.exp) return 'Unknown';
    
    const expirationDate = new Date(decoded.exp * 1000);
    return expirationDate.toLocaleString();
  } catch (error) {
    console.error('Error getting token expiration time:', error);
    return 'Unknown';
  }
};

/**
 * Get time remaining until token expires
 * @param {string} token - JWT token
 * @returns {number} - Minutes remaining or 0 if expired
 */
export const getTokenTimeRemaining = (token) => {
  try {
    const decoded = decodeToken(token);
    if (!decoded || !decoded.exp) return 0;
    
    const currentTime = Date.now() / 1000;
    const timeRemaining = decoded.exp - currentTime;
    
    return timeRemaining > 0 ? Math.floor(timeRemaining) : 0; // Return seconds
  } catch (error) {
    console.error('Error getting token time remaining:', error);
    return 0;
  }
};

/**
 * Format seconds as HH:MM:SS
 */
export const formatHMS = (totalSeconds) => {
  if (totalSeconds == null || Number.isNaN(totalSeconds)) return 'N/A';
  const secs = Math.max(0, Math.floor(totalSeconds));
  const h = Math.floor(secs / 3600);
  const m = Math.floor((secs % 3600) / 60);
  const s = secs % 60;
  const pad = (n) => n.toString().padStart(2, '0');
  return `${pad(h)}:${pad(m)}:${pad(s)}`;
};

  getCurrentUser,

  getTokenExpirationTime,
  getTokenTimeRemaining,
  formatHMS,
};
